package com.example.gestionstationskii.services;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import com.example.gestionstationskii.entities.*;
import com.example.gestionstationskii.repositories.*;

import org.springframework.transaction.annotation.Transactional;
import org.springframework.transaction.annotation.Isolation;
import javax.persistence.EntityManager;
import java.time.Period;
import java.util.List;
@RequiredArgsConstructor
@Service
@Transactional(isolation = Isolation.READ_COMMITTED)
public class RegistrationServicesImpl implements IRegistrationServices {

    private final IRegistrationRepository registrationRepository;
    private final ISkierRepository skierRepository;
    private final ICourseRepository courseRepository;
    private final PlanningSlotRepository slotRepository;
    private final IInstructorRepository instructorRepository;
    private final EntityManager entityManager;


    @Override
    public Registration addRegistrationAndAssignToSkier(Registration registration, Long numSkier) {
        Checks.require(registration.getNumRegistration() == null, "Identifiant généré à la création");
        Checks.require(registration.getPlanningSlot() != null, "Créneau requis ; utiliser /registration/skier/{id}/slot/{id}");
        return registerSlot(numSkier, registration.getPlanningSlot().getId());
    }

    @Override
    public Registration assignRegistrationToCourse(Long numRegistration, Long numCourse) {
        Checks.id(numRegistration); Checks.id(numCourse);
        Registration registration = registrationRepository.findById(numRegistration).orElseThrow(() -> BusinessException.missing("Inscription"));
        courseRepository.findById(numCourse).orElseThrow(() -> BusinessException.missing("Cours"));
        if (registration.isCancelled()) throw BusinessException.conflict("Inscription annulée");
        if (registration.getCourse() != null && registration.getCourse().getNumCourse().equals(numCourse)) return registration;
        throw BusinessException.conflict("Une inscription ne change pas de cours ; annuler puis réinscrire sur un créneau");
    }

    @Override
    public Registration addRegistrationAndAssignToSkierAndCourse(Registration registration, Long numSkieur, Long numCours) {
        Checks.id(numCours);
        courseRepository.findById(numCours).orElseThrow(() -> BusinessException.missing("Cours"));
        Checks.require(registration.getNumRegistration() == null, "Identifiant généré à la création");
        Checks.require(registration.getPlanningSlot() != null, "Créneau requis pour toute nouvelle inscription");
        Checks.id(registration.getPlanningSlot().getId());
        PlanningSlot slot = slotRepository.findById(registration.getPlanningSlot().getId()).orElseThrow(() -> BusinessException.missing("Créneau"));
        if (!slot.getCourse().getNumCourse().equals(numCours)) throw BusinessException.conflict("Créneau associé à un autre cours");
        return registerSlot(numSkieur, slot.getId());
    }

    @Override
    @Transactional(readOnly = true, isolation = Isolation.READ_COMMITTED)
    public List<Integer> numWeeksCourseOfInstructorBySupport(Long numInstructor, Support support) {
        Checks.id(numInstructor); Checks.require(support != null, "Support requis");
        instructorRepository.findById(numInstructor).orElseThrow(() -> BusinessException.missing("Moniteur"));
        return registrationRepository.numWeeksCourseOfInstructorBySupport(numInstructor, support);
    }

    @Override
    @Transactional(readOnly = true, isolation = Isolation.READ_COMMITTED)
    public PlanningSlot nextAvailableSlot(Long courseId) {
        return slotRepository.findAll().stream().filter(s -> s.getCourse().getNumCourse().equals(courseId)
                && s.getStatus() == SlotStatus.OPEN && s.getStartTime().isAfter(java.time.LocalDateTime.now()))
                .min(java.util.Comparator.comparing(PlanningSlot::getStartTime))
                .orElseThrow(() -> BusinessException.conflict("Aucun créneau disponible pour ce cours"));
    }

    @Override
    public Registration registerSlot(Long skierId, Long slotId) {
        Checks.id(skierId); Checks.id(slotId);
        // Serialize per skier (schedule conflicts), then per slot (capacity).
        Skier skier = skierRepository.lockById(skierId).orElseThrow(() -> BusinessException.missing("Skieur"));
        PlanningSlot slot = slotRepository.lockById(slotId).orElseThrow(() -> BusinessException.missing("Créneau"));
        entityManager.refresh(slot);
        if (slot.getStatus() != SlotStatus.OPEN || !slot.getStartTime().isAfter(java.time.LocalDateTime.now()))
            throw BusinessException.conflict("Créneau complet, annulé ou déjà commencé");
        Course course = slot.getCourse();
        Checks.require(skier.getDateOfBirth() != null, "Date de naissance du skieur requise");
        int age = Period.between(skier.getDateOfBirth(), slot.getStartTime().toLocalDate()).getYears();
        if ((course.getTypeCourse() == TypeCourse.COLLECTIVE_CHILDREN && age >= 16)
                || (course.getTypeCourse() == TypeCourse.COLLECTIVE_ADULT && age < 16))
            throw BusinessException.conflict("Âge incompatible avec le cours (seuil : 16 ans)");
        for (Registration r : registrationRepository.findBySkier_NumSkier(skierId)) {
            if (r.isCancelled()) continue;
            PlanningSlot other = r.getPlanningSlot();
            if (other != null && other.getStatus() != SlotStatus.CANCELLED
                    && TimeRanges.overlaps(other.getStartTime(), other.getEndTime(), slot.getStartTime(), slot.getEndTime()))
                throw BusinessException.conflict("Inscription en double ou conflit horaire du skieur");
            if (other != null && other.getStatus() != SlotStatus.CANCELLED && r.getCourse().getNumCourse().equals(course.getNumCourse())
                    && other.getStartTime().get(java.time.temporal.IsoFields.WEEK_BASED_YEAR) == slot.getStartTime().get(java.time.temporal.IsoFields.WEEK_BASED_YEAR)
                    && other.getStartTime().get(java.time.temporal.IsoFields.WEEK_OF_WEEK_BASED_YEAR) == slot.getStartTime().get(java.time.temporal.IsoFields.WEEK_OF_WEEK_BASED_YEAR))
                throw BusinessException.conflict("Skieur déjà inscrit à ce cours pendant cette semaine ISO");
            if (other == null && r.getCourse() != null && r.getCourse().getNumCourse().equals(course.getNumCourse())
                    && r.getNumWeek() == slot.getStartTime().get(java.time.temporal.IsoFields.WEEK_OF_WEEK_BASED_YEAR))
                throw BusinessException.conflict("Inscription historique déjà présente cette semaine (année inconnue)");
        }
        long count = registrationRepository.findByPlanningSlot_Id(slotId).stream().filter(r -> !r.isCancelled()).count();
        if (count >= slot.getMaxCapacity()) throw BusinessException.conflict("Capacité du créneau atteinte");
        Registration r = new Registration();
        r.setSkier(skier); r.setCourse(course); r.setPlanningSlot(slot);
        r.setNumWeek(slot.getStartTime().get(java.time.temporal.IsoFields.WEEK_OF_WEEK_BASED_YEAR));
        if (count + 1 >= slot.getMaxCapacity()) slot.setStatus(SlotStatus.FULL);
        return registrationRepository.save(r);
    }
    @Override
    public void cancel(Long id) {
        Checks.id(id);
        Registration initial = registrationRepository.findById(id).orElseThrow(() -> BusinessException.missing("Inscription"));
        skierRepository.lockById(initial.getSkier().getNumSkier()).orElseThrow(() -> BusinessException.missing("Skieur"));
        PlanningSlot slot = initial.getPlanningSlot() == null ? null : slotRepository.lockById(initial.getPlanningSlot().getId()).orElseThrow(() -> BusinessException.missing("Créneau"));
        entityManager.refresh(initial);
        if (slot != null) entityManager.refresh(slot);
        if (initial.isCancelled()) return;
        if (slot != null && !slot.getStartTime().isAfter(java.time.LocalDateTime.now())) throw BusinessException.conflict("Inscription commencée non annulable");
        initial.setCancelled(true);
        if (slot != null && slot.getStatus() == SlotStatus.FULL) slot.setStatus(SlotStatus.OPEN);
    }
    @Override
    @Transactional(readOnly = true, isolation = Isolation.READ_COMMITTED)
    public List<Registration> bySkier(Long id) {
        Checks.id(id); skierRepository.findById(id).orElseThrow(() -> BusinessException.missing("Skieur"));
        return registrationRepository.findBySkier_NumSkier(id);
    }
    @Override
    @Transactional(readOnly = true, isolation = Isolation.READ_COMMITTED)
    public List<Registration> search(java.time.LocalDateTime from, java.time.LocalDateTime to, Long courseId) {
        Checks.require(from != null && to != null && from.isBefore(to), "Période invalide");
        if (courseId != null) { Checks.id(courseId); courseRepository.findById(courseId).orElseThrow(() -> BusinessException.missing("Cours")); }
        return registrationRepository.findAll().stream().filter(r -> r.getPlanningSlot() != null
                && (courseId == null || r.getCourse().getNumCourse().equals(courseId))
                && TimeRanges.overlaps(r.getPlanningSlot().getStartTime(), r.getPlanningSlot().getEndTime(), from, to))
                .collect(java.util.stream.Collectors.toList());
    }

}

package com.example.gestionstationskii.services;

import com.example.gestionstationskii.dto.Requests;
import com.example.gestionstationskii.entities.*;
import com.example.gestionstationskii.repositories.*;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.transaction.annotation.Isolation;
import java.time.LocalDateTime;
import java.util.List;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Transactional(isolation = Isolation.READ_COMMITTED)
public class PlanningServicesImpl implements IPlanningServices {
    private final PlanningSlotRepository slots;
    private final InstructorAvailabilityRepository availabilities;
    private final IInstructorRepository instructors;
    private final ICourseRepository courses;
    private final IRegistrationRepository registrations;

    @Override
    public InstructorAvailability addAvailability(Long instructorId, Requests.Availability input) {
        Checks.id(instructorId);
        Instructor instructor = instructors.lockById(instructorId).orElseThrow(() -> BusinessException.missing("Moniteur"));
        Checks.require(input.getDayOfWeek() != null && input.getStartTime() != null && input.getEndTime() != null
                && input.getStartTime().isBefore(input.getEndTime()), "Jour et heures valides requis (sans passage à minuit)");
        Checks.require(input.getValidFrom() != null && input.getValidUntil() != null && !input.getValidFrom().isAfter(input.getValidUntil()), "Période de validité invalide");
        InstructorAvailability a = new InstructorAvailability();
        a.setInstructor(instructor); a.setDayOfWeek(input.getDayOfWeek()); a.setStartTime(input.getStartTime()); a.setEndTime(input.getEndTime());
        a.setValidFrom(input.getValidFrom()); a.setValidUntil(input.getValidUntil());
        return availabilities.save(a);
    }
    @Override
    @Transactional(readOnly = true, isolation = Isolation.READ_COMMITTED)
    public List<InstructorAvailability> availability(Long id) {
        Checks.id(id); instructors.findById(id).orElseThrow(() -> BusinessException.missing("Moniteur"));
        return availabilities.findByInstructor_NumInstructor(id);
    }
    @Override
    @Transactional(readOnly = true, isolation = Isolation.READ_COMMITTED)
    public PlanningSlot get(Long id) {
        Checks.id(id); return slots.findById(id).orElseThrow(() -> BusinessException.missing("Créneau"));
    }
    @Override
    public PlanningSlot create(Requests.Slot input) { return save(new PlanningSlot(), input); }
    @Override
    public PlanningSlot update(Long id, Requests.Slot input) {
        Checks.id(id);
        PlanningSlot current = slots.lockById(id).orElseThrow(() -> BusinessException.missing("Créneau"));
        if (current.getStatus() == SlotStatus.CANCELLED) throw BusinessException.conflict("Créneau annulé non modifiable");
        if (!current.getStartTime().isAfter(LocalDateTime.now())) throw BusinessException.conflict("Créneau commencé non modifiable");
        // Do not silently move a skier's confirmed appointment.
        if (!registrations.findByPlanningSlot_Id(id).isEmpty()) throw BusinessException.conflict("Créneau avec historique d'inscriptions non modifiable");
        return save(current, input);
    }
    private PlanningSlot save(PlanningSlot slot, Requests.Slot input) {
        Checks.id(input.getCourseId()); Checks.id(input.getInstructorId());
        Checks.require(input.getStartTime() != null && input.getEndTime() != null && input.getStartTime().isBefore(input.getEndTime())
                && input.getStartTime().isAfter(LocalDateTime.now()) && input.getStartTime().toLocalDate().equals(input.getEndTime().toLocalDate()), "Créneau futur, de durée positive et sur une même journée requis");
        Course course = courses.lockById(input.getCourseId()).orElseThrow(() -> BusinessException.missing("Cours"));
        Instructor instructor = instructors.lockById(input.getInstructorId()).orElseThrow(() -> BusinessException.missing("Moniteur"));
        if (!instructor.getCourses().contains(course)) throw BusinessException.conflict("Ce moniteur n'est pas associé au cours");
        int ceiling = course.getTypeCourse() == TypeCourse.INDIVIDUAL ? 1 : 6;
        Checks.require(input.getMaxCapacity() != null && input.getMaxCapacity() > 0 && input.getMaxCapacity() <= ceiling,
                "Capacité requise : 1 en individuel, au plus 6 en collectif");
        boolean covered = availability(instructor.getNumInstructor()).stream().anyMatch(a -> a.getDayOfWeek() == input.getStartTime().getDayOfWeek()
                && !input.getStartTime().toLocalDate().isBefore(a.getValidFrom()) && !input.getEndTime().toLocalDate().isAfter(a.getValidUntil())
                && !input.getStartTime().toLocalTime().isBefore(a.getStartTime()) && !input.getEndTime().toLocalTime().isAfter(a.getEndTime()));
        if (!covered) throw BusinessException.conflict("Créneau hors des disponibilités du moniteur");
        boolean overlap = slots.findAll().stream().anyMatch(s -> !s.getId().equals(slot.getId()) && s.getStatus() != SlotStatus.CANCELLED
                && s.getInstructor().getNumInstructor().equals(input.getInstructorId())
                && TimeRanges.overlaps(s.getStartTime(), s.getEndTime(), input.getStartTime(), input.getEndTime()));
        if (overlap) throw BusinessException.conflict("Le moniteur a déjà un cours sur ce créneau");
        slot.setCourse(course); slot.setInstructor(instructor); slot.setStartTime(input.getStartTime()); slot.setEndTime(input.getEndTime());
        slot.setMaxCapacity(input.getMaxCapacity()); slot.setStatus(SlotStatus.OPEN);
        return slots.save(slot);
    }
    @Override
    public void cancel(Long id) {
        Checks.id(id);
        PlanningSlot slot = slots.lockById(id).orElseThrow(() -> BusinessException.missing("Créneau"));
        if (!slot.getStartTime().isAfter(LocalDateTime.now())) throw BusinessException.conflict("Créneau commencé non annulable");
        slot.setStatus(SlotStatus.CANCELLED);
        registrations.findByPlanningSlot_Id(id).forEach(r -> r.setCancelled(true));
    }
    @Override
    @Transactional(readOnly = true, isolation = Isolation.READ_COMMITTED)
    public List<PlanningSlot> search(LocalDateTime from, LocalDateTime to, Long courseId, Long instructorId, boolean available) {
        Checks.require(from != null && to != null && from.isBefore(to), "Période de recherche invalide");
        if (courseId != null) { Checks.id(courseId); courses.findById(courseId).orElseThrow(() -> BusinessException.missing("Cours")); }
        if (instructorId != null) { Checks.id(instructorId); instructors.findById(instructorId).orElseThrow(() -> BusinessException.missing("Moniteur")); }
        return slots.findAll().stream().filter(s -> TimeRanges.overlaps(s.getStartTime(), s.getEndTime(), from, to)
                && (courseId == null || s.getCourse().getNumCourse().equals(courseId))
                && (instructorId == null || s.getInstructor().getNumInstructor().equals(instructorId))
                && (!available || (s.getStatus() == SlotStatus.OPEN && s.getStartTime().isAfter(LocalDateTime.now()))))
                .sorted(java.util.Comparator.comparing(PlanningSlot::getStartTime)).collect(Collectors.toList());
    }
}
